```
# Development of Python Code Compatible with Multiple AI Tools

import requests
import json

# -------------------------------
# API Keys (Replace with your keys)
# -------------------------------

OPENAI_API_KEY = "your_openai_api_key"
GEMINI_API_KEY = "your_gemini_api_key"
HF_API_KEY = "your_huggingface_api_key"

# -------------------------------
# Prompt for AI Tools
# -------------------------------

prompt = """
Analyze the following student marks:
Math = 85
Science = 78
English = 92

Provide performance insights and suggestions for improvement.
"""

# -------------------------------
# OpenAI API
# -------------------------------

def get_openai_response(prompt):

    url = "https://api.openai.com/v1/chat/completions"

    headers = {
        "Authorization": f"Bearer {OPENAI_API_KEY}",
        "Content-Type": "application/json"
    }

    data = {
        "model": "gpt-4o-mini",
        "messages": [
            {"role": "user", "content": prompt}
        ]
    }

    response = requests.post(url, headers=headers, json=data)

    return response.json()

# -------------------------------
# Gemini API
# -------------------------------

def get_gemini_response(prompt):

    url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent?key={GEMINI_API_KEY}"

    headers = {
        "Content-Type": "application/json"
    }

    data = {
        "contents": [
            {
                "parts": [
                    {"text": prompt}
                ]
            }
        ]
    }

    response = requests.post(url, headers=headers, json=data)

    return response.json()

# -------------------------------
# Hugging Face API
# -------------------------------

def get_huggingface_response(prompt):

    API_URL = "https://api-inference.huggingface.co/models/gpt2"

    headers = {
        "Authorization": f"Bearer {HF_API_KEY}"
    }

    data = {
        "inputs": prompt
    }

    response = requests.post(API_URL, headers=headers, json=data)

    return response.json()

# -------------------------------
# Main Program
# -------------------------------

print("Generating AI Responses...\n")

# OpenAI Response
openai_output = get_openai_response(prompt)
print("OPENAI OUTPUT:")
print(json.dumps(openai_output, indent=2))

print("\n-----------------------------------\n")

# Gemini Response
gemini_output = get_gemini_response(prompt)
print("GEMINI OUTPUT:")
print(json.dumps(gemini_output, indent=2))

print("\n-----------------------------------\n")

# Hugging Face Response
hf_output = get_huggingface_response(prompt)
print("HUGGING FACE OUTPUT:")
print(json.dumps(hf_output, indent=2))
```

The program imports the required Python libraries such as requests and json.
